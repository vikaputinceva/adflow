<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class AuthController extends Controller
{
    public function showLogin()
    {
        return view('login');
    }

    public function login(Request $request)
    {
        $data = $request->validate([
            'email' => 'required|email',
            'password' => 'required',
        ]);

        if (Auth::attempt($data)) {
            $request->session()->regenerate();
            return redirect('/');
        }

        return back()->withErrors(['email' => 'РќРµРІРµСЂРЅС‹Р№ email РёР»Рё РїР°СЂРѕР»СЊ'])->withInput();
    }

    public function logout(Request $request)
    {
        Auth::logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();
        return redirect('/login');
    }
}

namespace App\Http\Controllers;

use App\Models\Campaign;
use App\Models\CampaignMaterial;
use App\Models\Platform;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

class CampaignController extends Controller
{
    public function index(Request $request)
    {
        $status = $request->get('status', 'all');
        $search = trim((string) $request->get('search', ''));

        $query = Campaign::query();

        if ($status !== 'all') {
            $query->where('status', $status);
        }

        if ($search !== '') {
            $query->where(function ($q) use ($search) {
                $q->where('name', 'like', '%' . $search . '%')
                  ->orWhere('platform_name', 'like', '%' . $search . '%');
            });
        }

        $campaigns = $query->orderBy('id', 'desc')->get();

        // РІС‹Р±СЂР°РЅРЅР°СЏ РєР°РјРїР°РЅРёСЏ (РїРѕ СѓРјРѕР»С‡Р°РЅРёСЋ РїРµСЂРІР°СЏ РІ СЃРїРёСЃРєРµ)
        $selectedCampaign = $campaigns->firstWhere('id', $request->get('selected')) ?? $campaigns->first();

        $materials = $selectedCampaign
            ? $selectedCampaign->materials()->orderBy('id', 'desc')->get()
            : collect();

        $totals = Campaign::totals();

        return view('campaigns.index', [
            'campaigns' => $campaigns,
            'platforms' => Platform::orderBy('name')->get(),
            'selectedCampaign' => $selectedCampaign,
            'materials' => $materials,
            'currentStatus' => $status,
            'search' => $search,
            'budgetUsedPercent' => $totals['used_percent'],
            'budgetLeftPercent' => $totals['left_percent'],
        ]);
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'name' => 'required|string|max:200',
            'platform_name' => 'required|exists:platforms,name',
            'budget_total' => 'required|numeric|min:1',
            'end_date' => 'required|date',
        ]);

        Campaign::create($data + [
            'status' => 'active',
            'budget_spent' => 0,
            'revenue_attributed' => 0,
        ]);

        return redirect('/campaigns')->with('success', 'РљР°РјРїР°РЅРёСЏ РґРѕР±Р°РІР»РµРЅР°');
    }

    public function updateStatus(Request $request, Campaign $campaign)
    {
        $data = $request->validate([
            'status' => 'required|in:active,paused,completed',
        ]);

        $campaign->update($data);

        return redirect('/campaigns?selected=' . $campaign->id)->with('success', 'РЎС‚Р°С‚СѓСЃ РѕР±РЅРѕРІР»С‘РЅ');

        return round($this->budget_spent / $this->budget_total * 100);
    }

    {
        return max(0, $this->budget_total - $this->budget_spent);

        return round(($this->revenue_attributed - $this->budget_spent) / $this->budget_spent * 100);
    }

namespace App\Services;

final class TelegramNotifier
{
    public function __construct(
        private readonly string $botToken,
        private readonly string $defaultChatId
    ) {}

    public function sendMessage(string $text, ?string $chatId = null): bool
    {
        $targetChatId = $chatId ?: $this->defaultChatId;
        if ($this->botToken === '' || $targetChatId === '') {
            return false;
        }

        $url = sprintf('https://api.telegram.org/bot%s/sendMessage', $this->botToken);
        $payload = json_encode([
            'chat_id' => $targetChatId,
            'text' => $text,
        ], JSON_UNESCAPED_UNICODE);

        $ch = curl_init($url);
        curl_setopt_array($ch, [
            CURLOPT_POST => true,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_HTTPHEADER => ['Content-Type: application/json'],
            CURLOPT_POSTFIELDS => $payload,
            CURLOPT_TIMEOUT => 5,
            CURLOPT_SSL_VERIFYPEER => false,
            CURLOPT_SSL_VERIFYHOST => 0,
        ]);

        $response = curl_exec($ch);
        $httpCode = (int) curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);

        if ($response === false || $httpCode !== 200) {
            return false;
        }

        $decoded = json_decode((string) $response, true);
        return is_array($decoded) && ($decoded['ok'] ?? false) === true;
    }

    public function notifyDeadline(string $taskTitle, string $deadline, ?string $chatId = null): bool
    {
        return $this->sendMessage(

            $chatId
        );
    }
